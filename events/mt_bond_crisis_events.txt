add_namespace = mt_bond_crisis

# Sovereign debt crisis chain, triggered by a GM targeting a country with the
# d_mt_gm_bond_crisis decision (common/decisions/mt_gc_tools_decisions.txt).
#
# Severity is gated on debt_ratio (debt/GDP, already computed weekly in calculate_interest_rate,
# 00_money_system.txt): a country already deep in debt takes the worse outcome.

news_event = {
	id = mt_bond_crisis.1
	picture = GFX_news_event_bond_crisis_1

	title = mt_bond_crisis.1.t
	desc = mt_bond_crisis.1.d
	major = yes
	is_triggered_only = yes

	option = {
		name = mt_bond_crisis.1.o1

		add_timed_idea = { idea = mt_bond_crisis days = 1080 }
		decrease_economic_growth = yes

		hidden_effect = {
			country_event = {
				id = mt_bond_crisis.2
				days = 7
			}
		}
	}
}

news_event = {
	id = mt_bond_crisis.2
	picture = GFX_news_event_bond_crisis_2

	title = mt_bond_crisis.2.t
	desc = mt_bond_crisis.2.d
	is_triggered_only = yes

	option = {
		name = mt_bond_crisis.2.o1

		trigger = {
			check_variable = { var = debt_ratio value = 1.0 compare = greater_than_or_equals }
		}

		add_ideas = mt_currency_devaluation
		decrease_economic_growth = yes

		hidden_effect = {
			country_event = {
				id = mt_bond_crisis.3
				days = 7
			}
		}
	}

	option = {
		name = mt_bond_crisis.2.o1

		trigger = {
			check_variable = { debt_ratio < 1.0 }
		}

		add_ideas = mt_currency_devaluation_min
		decrease_economic_growth = yes

		hidden_effect = {
			country_event = {
				id = mt_bond_crisis.3
				days = 7
			}
		}
	}
}
