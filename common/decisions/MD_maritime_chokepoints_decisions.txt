	MD_suez_close_canal = {
		icon = GFX_decision_israel_blockade
		on_map_mode = map_and_decisions_view
		highlight_states = {
			highlight_state_targets = {
				state = 213
				state = 214
			}
		}
		visible = {
			OR = {
				controls_state = 213
				controls_state = 214
			}
			NOT = { has_global_flag = SUEZ_CANAL_BLOCKED }
		}
		available = { is_ai = no }
		cost = 150
		complete_effect = {
			set_global_flag = SUEZ_CANAL_BLOCKED
			add_ideas = MD_suez_closure_backlash_idea
			add_threat = 10
			every_other_country = {
				limit = { NOT = { is_in_faction_with = ROOT } }
				add_ideas = MD_suez_denied_idea
			}
			# Faction members are automatically exempted (tracked the same way a manual
			# MD_suez_exempt_grant would be, so MD_suez_exempt_revoke can still individually
			# strip it from one of them later if wanted).
			if = {
				limit = { is_in_faction = yes }
				every_faction_member = {
					limit = { NOT = { tag = ROOT } }
					set_country_flag = SUEZ_CANAL_EXEMPT_FOR_COUNTRY
					add_to_array = { array = MD_suez_exempt_by_them value = ROOT }
					ROOT = {
						add_to_array = { array = MD_suez_exempt_by_us value = PREV }
					}
				}
			}
			# See MD_maritime_chokepoints.2 (events/MD_maritime_chokepoints.txt): fires in 10
			# days and grants a wargoal to any co-controller, superpower, or regional power still
			# affected and not exempted by then.
			country_event = { id = MD_maritime_chokepoints.2 days = 10 }
			custom_effect_tooltip = MD_suez_closed_tt
		}
		ai_will_do = { base = 0 }
	}

	### STRAIT OF HORMUZ (PER / OMA) ###
	# Relocated from common/decisions/Iran.txt (was filed under the "Iranian Domestic
	# Affairs" category, PER_domestic_politics) so all five chokepoints live in one tab.
	# Converted from a 12-day delayed decision (remove_effect/days_remove) to an instant
	# one (complete_effect/cost) to match every other chokepoint closure, and given the
	# cost = 150 it was missing - both fixed after being reported as inconsistent.

	PER_block_hormuz = {
		icon = GFX_decision_israel_blockade
		on_map_mode = map_and_decisions_view
		highlight_states = {
			highlight_state_targets = {
				state = 978
				state = 425
				state = 1119
			}
		}
		available = {
			is_ai = no
		}
		visible = {
			OR = {
				controls_state = 978
				controls_state = 425
				controls_state = 1119
			}
			NOT = {
				has_global_flag = HORMUZ_BLOCKED
			}
		}
		cost = 150
		complete_effect = {
			log = "[GetDateText]: [Root.GetName]: Decision PER_block_hormuz"
			set_global_flag = HORMUZ_BLOCKED
			add_ideas = MD_hormuz_closure_backlash_idea
			# Tracks who specifically closed it, since PER_hormuz_blocked_gang's cancel needs to
			# exempt the initiator, not just "anyone who controls a Hormuz state" - Iran and Oman
			# each control a different state, so mere control no longer implies "did the blocking"
			# now that either can invoke this decision.
			set_country_flag = HORMUZ_BLOCKADE_INITIATOR
			# Faction members are automatically exempted (tracked the same way a manual
			# MD_hormuz_exempt_grant would be, so MD_hormuz_exempt_revoke can still individually
			# strip it from one of them later if wanted). PER_hormuz_blocked_gang's own cancel
			# condition already respects HORMUZ_EXEMPT_FOR_COUNTRY, so this alone is enough to
			# keep the idea off exempted faction members regardless of the Gulf-state gate below.
			if = {
				limit = { is_in_faction = yes }
				every_faction_member = {
					limit = { NOT = { tag = ROOT } }
					set_country_flag = HORMUZ_EXEMPT_FOR_COUNTRY
					add_to_array = { array = MD_hormuz_exempt_by_them value = ROOT }
					ROOT = {
						add_to_array = { array = MD_hormuz_exempt_by_us value = PREV }
					}
				}
			}
			# See MD_maritime_chokepoints.2 (events/MD_maritime_chokepoints.txt): fires in 10
			# days and grants a wargoal to any co-controller, Gulf state, superpower, or regional
			# power still affected and not exempted by then.
			country_event = { id = MD_maritime_chokepoints.2 days = 10 }
			news_event = iranian_events.910
			add_threat = 10
			newline = yes
			#An immediate spike only: recalculate_resources_price rebuilds oil_price from supply/demand at the
			#next monthly tick. The lasting rise comes from PER_hormuz_blocked_gang's export penalty.
			set_temp_variable = {
				new_price = {
					value = global.oil_price
					multiply = 1.15
				}
				tooltip = oil_new_price_tt
			}
			set_variable = { global.oil_price = new_price }
		}

		ai_will_do = {
			base = 0
			modifier = {
				has_war_with = USA
				add = 10
			}
		}
	}
