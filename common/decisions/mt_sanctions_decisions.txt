# Named so the id matches what each decision actually displays (check_sanctions_imposed_by_us
# lists who we sanction, check_sanctions_imposed_on_us lists who sanctions us).
mt_sanctions_decisions = {

	# Lists every country we currently sanction (mt_sanctions.1, events/mt_sanctions_events.txt).
	d_mt_check_sanctions_imposed_by_us = {
		icon = GFX_decision_economy

		visible = {
			is_ai = no
		}

		available = {
			OR = {
				NOT = { check_variable = { mt_fin_sanc_by_us^0 = 0 } }
				NOT = { check_variable = { mt_eco_sanc_by_us^0 = 0 } }
				NOT = { check_variable = { mt_dip_sanc_by_us^0 = 0 } }
				NOT = { check_variable = { mt_mil_sanc_by_us^0 = 0 } }
			}
		}

		complete_effect = {
			country_event = mt_sanctions.1
		}
	}

	# Lists every country currently sanctioning us (mt_sanctions.2).
	d_mt_check_sanctions_imposed_on_us = {
		icon = GFX_decision_economy

		visible = {
			is_ai = no
		}

		available = {
			OR = {
				NOT = { check_variable = { mt_fin_sanc_on_us^0 = 0 } }
				NOT = { check_variable = { mt_eco_sanc_on_us^0 = 0 } }
				NOT = { check_variable = { mt_dip_sanc_on_us^0 = 0 } }
				NOT = { check_variable = { mt_mil_sanc_on_us^0 = 0 } }
			}
		}

		complete_effect = {
			country_event = mt_sanctions.2
		}
	}

	# Corruption-for-stability relief valve once any sanction category exceeds 10%.
	# The sanctions percentages are clamped [0,1] (a fraction), so the threshold below must be
	# 0.10, not 10.
	d_mt_consolidate_power = {
		icon = GFX_decision_economy
		days_remove = 90

		visible = {
			is_ai = no
			OR = {
				check_variable = { mt_dip_sanctions_p > 0.10 }
				check_variable = { mt_mil_sanctions_p > 0.10 }
				check_variable = { mt_eco_sanctions_p > 0.10 }
				check_variable = { mt_fin_sanctions_p > 0.10 }
			}
		}

		available = {
			NOT = { has_idea = corruption_level_10 }
		}

		remove_effect = {
			increase_corruption = yes
			add_political_power = 300

			if = {
				limit = { has_government = fascism }
				add_popularity = { ideology = fascism popularity = 0.05 }
			}
			if = {
				limit = { has_government = communism }
				add_popularity = { ideology = communism popularity = 0.05 }
			}
			if = {
				limit = { has_government = nationalist }
				add_popularity = { ideology = nationalist popularity = 0.05 }
			}
			if = {
				limit = { has_government = neutrality }
				add_popularity = { ideology = neutrality popularity = 0.05 }
			}
		}
	}
}
