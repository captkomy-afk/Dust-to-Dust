mt_bond_crisis_decisions = {

	# "Emergency measures" - 30 days after taking this, the government quietly skims part of
	# the crisis response: political power and treasury both go up, at the cost of corruption.
	d_mt_bond_crisis_1 = {
		icon = GFX_decision_economy

		days_remove = 30
		fire_only_once = yes

		visible = {
			has_idea = mt_bond_crisis
		}

		available = {
			NOT = { has_idea = corruption_level_10 }
		}

		remove_effect = {
			increase_corruption = yes
			add_political_power = 200

			set_temp_variable = { treasury_change = gdp_total }
			multiply_temp_variable = { treasury_change = 0.025 }
			modify_treasury_effect = yes
		}
	}

	# Cut interest rates to ease the crisis, at the cost of political power output.
	d_mt_bond_crisis_2 = {
		icon = GFX_decision_economy

		cost = 250
		days_remove = 120
		fire_only_once = yes

		visible = {
			has_idea = mt_bond_crisis
		}

		modifier = {
			interest_rate_multiplier_modifier = -1
			political_power_factor = -0.2
		}

		remove_effect = {
			modify_timed_idea = {
				idea = mt_bond_crisis
				days = -125
			}
		}
	}

	# Emergency bailout package: shortens the crisis at the cost of a 3.5%-of-GDP treasury hit.
	d_mt_bond_crisis_3 = {
		icon = GFX_decision_economy

		cost = 250
		days_remove = 120
		fire_only_once = yes

		visible = {
			has_idea = mt_bond_crisis
		}

		complete_effect = {
			set_temp_variable = { treasury_change = gdp_total }
			multiply_temp_variable = { treasury_change = -0.035 }
			modify_treasury_effect = yes
		}

		remove_effect = {
			modify_timed_idea = {
				idea = mt_bond_crisis
				days = -125
			}
		}
	}

		}

		modifier = {
			production_speed_buildings_factor = -0.5

			modify_timed_idea = {
				idea = mt_bond_crisis
				days = -125
			}
		}
	}
