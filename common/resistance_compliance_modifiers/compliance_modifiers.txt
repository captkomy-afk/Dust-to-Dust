		if = {
			limit = {
				OR = {
					NOT = { original_tag = CHI }

					# in the modifier's own State scope (compliance modifiers are inherently per-state);
					# wrapping it in "FROM" switched to Country scope instead, which made the whole
					# check invalid every time ("Invalid Scope, supported: State, provided: Country").
					# Removed the FROM wrapper so the state checks run directly against this state.
					NOT = { state = 592 }
					NOT = { state = 589 }
					NOT = { state = 556 }
				}
			}
			# occupied_countries.1 has no localisation entries anywhere - firing it shows a
			# blank/raw-key popup. Commented out until loc is written; re-enable by uncommenting.
			# country_event = occupied_countries.1
		}
