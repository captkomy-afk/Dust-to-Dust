			# Reprice active contracts that explicitly follow the market.

			# One-time migration for old saves: auto purchase starts disabled, then stays player-controlled.

				limit = {
					is_ai = no

				}
				clr_country_flag = dtd_enable_auto_buy_cflag
				set_country_flag = dtd_auto_buy_default_initialized_cflag
			}

			# CKEdited - dtd_daily_effect / dtd_update_trade_opinion_attractiveness_cache_effect are not

			#dtd_update_trade_opinion_attractiveness_cache_effect = yes

				count = modifier@resource_trade_cost_bonus_per_factory

				count = 8
			}
			set_variable = {

			}
