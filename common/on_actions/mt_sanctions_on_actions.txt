# Sanctions calculation cadence and cleanup. No starting-sanctions seed is set up here - that
# would need to be scenario/country-roster specific and is left to the modder to configure.
# NOTE: sanctions won't be properly calculated until the second month, since the global
# GDP/office/factory denominators need one full every_country pass first.

on_actions = {
	## Extra Economic Calculations. Requires to calculate sanction strength.
	on_monthly_LIC = {
		effect = {
			if = {
				# Calculate the total office parks and mil/docks. We only want to do this once
				# per month globally, not once per country - the control scope (Liechtenstein)
				# cannot be annexed, so it's a safe place to gate a once-per-month reset.
				set_variable = { global.mt_global_num_offices = 0 }
				set_variable = { global.mt_global_num_mils = 0 }
				set_variable = { global.mt_world_gdp_total = 0 }
				every_country = {
					mt_sanctions_extra_calculations = yes
				}
			}

			## Now calculate outgoing sanction strength
			every_country = {
				mt_sanctions_strength_calculate = yes
			}

			## Now calculate incoming sanction strength and blowback
			every_country = {
				mt_sanctions_calculate = yes
			}
		}
	}

	# Prune a defeated/annexed country out of every sanctions array so they don't linger forever.
	on_annex = {
		effect = {
			FROM = {
				clear_array = mt_fin_sanc_by_us
				clear_array = mt_eco_sanc_by_us
				clear_array = mt_dip_sanc_by_us
				clear_array = mt_mil_sanc_by_us
				clear_array = mt_fin_sanc_on_us
				clear_array = mt_eco_sanc_on_us
				clear_array = mt_dip_sanc_on_us
				clear_array = mt_mil_sanc_on_us
			}

			every_country = {
				if = {
					limit = { is_in_array = { array = mt_fin_sanc_by_us value = FROM } }
					remove_from_array = { mt_fin_sanc_by_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_eco_sanc_by_us value = FROM } }
					remove_from_array = { mt_eco_sanc_by_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_dip_sanc_by_us value = FROM } }
					remove_from_array = { mt_dip_sanc_by_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_mil_sanc_by_us value = FROM } }
					remove_from_array = { mt_mil_sanc_by_us = FROM }
				}

				if = {
					limit = { is_in_array = { array = mt_fin_sanc_on_us value = FROM } }
					remove_from_array = { mt_fin_sanc_on_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_eco_sanc_on_us value = FROM } }
					remove_from_array = { mt_eco_sanc_on_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_dip_sanc_on_us value = FROM } }
					remove_from_array = { mt_dip_sanc_on_us = FROM }
				}
				if = {
					limit = { is_in_array = { array = mt_mil_sanc_on_us value = FROM } }
					remove_from_array = { mt_mil_sanc_on_us = FROM }
				}
			}
		}
	}

	# Threshold-gated flavor events: 25% aggregate sanction level, 90-day global cooldown plus a
	# 360-day per-category cooldown (180 days for blowback).
	on_monthly = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = mt_sanction_event_cd }
				}

				if = {
					limit = {
						check_variable = { mt_dip_sanctions_p > 0.25 }
						NOT = { has_country_flag = mt_dip_sanction_event_cd }
					}
					set_country_flag = { flag = mt_sanction_event_cd days = 90 value = 1 }
					set_country_flag = { flag = mt_dip_sanction_event_cd days = 360 value = 1 }
					mt_dip_event = yes
				}
				else_if = {
					limit = {
						check_variable = { mt_eco_sanctions_p > 0.25 }
						NOT = { has_country_flag = mt_eco_sanction_event_cd }
					}
					set_country_flag = { flag = mt_sanction_event_cd days = 90 value = 1 }
					set_country_flag = { flag = mt_eco_sanction_event_cd days = 360 value = 1 }
					mt_eco_event = yes
				}
				else_if = {
					limit = {
						check_variable = { mt_fin_sanctions_p > 0.25 }
						NOT = { has_country_flag = mt_fin_sanction_event_cd }
					}
					set_country_flag = { flag = mt_sanction_event_cd days = 90 value = 1 }
					set_country_flag = { flag = mt_fin_sanction_event_cd days = 360 value = 1 }
					mt_fin_event = yes
				}
				else_if = {
					limit = {
						check_variable = { mt_mil_sanctions_p > 0.25 }
						NOT = { has_country_flag = mt_mil_sanction_event_cd }
					}
					set_country_flag = { flag = mt_sanction_event_cd days = 90 value = 1 }
					set_country_flag = { flag = mt_mil_sanction_event_cd days = 360 value = 1 }
					mt_mil_event = yes
				}

				if = {
					limit = {
						check_variable = { mt_sanctions_blowback_p > 0.25 }
						NOT = { has_country_flag = mt_bb_sanction_event_cd }
					}
					set_country_flag = { flag = mt_bb_sanction_event_cd days = 180 value = 1 }
					mt_bb_event = yes
				}
			}
		}
	}
}
