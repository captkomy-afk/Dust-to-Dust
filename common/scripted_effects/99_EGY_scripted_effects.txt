EGY_update_military_standing = {
	set_variable = { EGY_military_standing_value = 0 }

	#Military spending
	set_variable = { EGY_military_standing_spending = 0 }
	if = {
		limit = { has_idea = defence_04 }
		add_to_variable = { EGY_military_standing_spending = 20 }
	}
	else_if = {
		limit = { has_idea = defence_05 }
		add_to_variable = { EGY_military_standing_spending = 40 }
	}
	else_if = {
		limit = { has_idea = defence_06 }
		add_to_variable = { EGY_military_standing_spending = 60 }
	}
	else_if = {
		limit = { has_idea = defence_07 }
		add_to_variable = { EGY_military_standing_spending = 80 }
	}
	else_if = {
		limit = { has_idea = defence_08 }
		add_to_variable = { EGY_military_standing_spending = 100 }
	}
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_spending }

	#Military internal faction opinion counts as positive
	set_variable = { EGY_military_standing_opinion = ROOT.the_military_opinion }
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_opinion }

	#Corruption above level 5 is -20 points per level
	set_variable = { EGY_military_standing_corruption = 0 }
	if = {
		limit = { has_idea = corruption_level_06 }
		add_to_variable = { EGY_military_standing_corruption = 20 }
	}
	else_if = {
		limit = { has_idea = corruption_level_07 }
		add_to_variable = { EGY_military_standing_corruption = 40 }
	}
	else_if = {
		limit = { has_idea = corruption_level_08 }
		add_to_variable = { EGY_military_standing_corruption = 60 }
	}
	else_if = {
		limit = { has_idea = corruption_level_09 }
		add_to_variable = { EGY_military_standing_corruption = 80 }
	}
	else_if = {
		limit = { has_idea = corruption_level_10 }
		add_to_variable = { EGY_military_standing_corruption = 100 }
	}
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_corruption }

	#Bad economic cycle is -25 points per level below stable growth
	set_variable = { EGY_military_standing_economy = 0 }
	if = {
		limit = { has_idea = stagnation }
		add_to_variable = { EGY_military_standing_economy = -25 }
	}
	else_if = {
		limit = { has_idea = recession }
		add_to_variable = { EGY_military_standing_economy = -50 }
	}
	else_if = {
		limit = { has_idea = depression }
		add_to_variable = { EGY_military_standing_economy = -75 }
	}
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_economy }

	#Interest rate multiplied by -5
	set_variable = { EGY_military_standing_interest_rate = ROOT.interest_rate }
	round_variable = EGY_military_standing_interest_rate
	multiply_variable = { EGY_military_standing_interest_rate = -5 }
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_interest_rate }

	#Different parties have different base opinions
	set_variable = { EGY_military_standing_party = 0 }
	if = { # Muslim Brotherhood, Coptic, & Shi'ites
		limit = {
			OR = {
				is_in_array = { ruling_party = 12 }
				is_in_array = { ruling_party = 9 }
				is_in_array = { ruling_party = 8 }
			}
		}
		set_variable = { EGY_military_standing_party = -100 }
	}
	else_if = {
		limit = { # Salafists
			OR = {
				is_in_array = { ruling_party = 10 }
				is_in_array = { ruling_party = 11 }
			}
		}
		set_variable = { EGY_military_standing_party = -50 }
	}
	else_if = {
		limit = {
			OR = {
				is_in_array = { ruling_party = 0 }
				is_in_array = { ruling_party = 1 }
				is_in_array = { ruling_party = 2 }
				is_in_array = { ruling_party = 3 }
				is_in_array = { ruling_party = 7 }
				is_in_array = { ruling_party = 21 }
			}
		}
		set_variable = { EGY_military_standing_party = 50 }
	}
	else_if = {
		limit = { is_in_array = { ruling_party = 22 } }
		set_variable = { EGY_military_standing_party = 100 }
	}
	add_to_variable = { EGY_military_standing_value = EGY_military_standing_party }
}

# CKEdited - Coptic party political mechanic rework.
# The Coptic party (party_pop_array^13, generic type "Neutral_Autocracy" reflavored for EGY as
# "Coptic Christian Authority") previously grew/shrank purely through the generic MD politics engine's
# drift-based redistribution, completely decoupled from EGY_coptic_people_opinion (the satisfaction ladder
# that only drove national spirit modifiers). This ties the two together:
#  - Satisfied Copts hand a majority of their political weight to the ruling party (if it isn't Islamist);
#    unsatisfied/enraged Copts radicalize into the Coptic party itself instead.
#  - If the ruling party is Islamist (Freedom and Justice/Muslim Brotherhood, Hazemoon, Salafist Kingdom/
#    Caliphate, or the Shiite Vilayat_e_Faqih variants), it never benefits - its would-be share is instead
#    spread across the leading party in each other non-Islamist outlook.
#  - Natural growth of the Coptic party is capped at state 217 (Nile Valley)'s share of Egypt's total owned
#    state population. Enraged-tier radicalization can exceed the cap; manual PP-funded intervention via the
#    EGY_promote_coptic_politicians decision can too.

EGY_calculate_coptic_pop_cap = { # CKEdited
	set_variable = { EGY_coptic_total_state_pop = 0 }
	every_owned_state = {
		ROOT = {
			add_to_variable = { EGY_coptic_total_state_pop = PREV.state_population_k }
		}
	}
	set_variable = { EGY_coptic_state_pop = 0 }
	if = {
		limit = { owns_state = 217 }
		217 = {
			ROOT = {
				set_variable = { EGY_coptic_state_pop = PREV.state_population_k }
			}
		}
	}
	set_variable = { EGY_coptic_pop_cap = EGY_coptic_state_pop }
	if = {
		limit = { check_variable = { EGY_coptic_total_state_pop > 0 } }
		divide_variable = { EGY_coptic_pop_cap = EGY_coptic_total_state_pop }
	}
	clamp_variable = { var = EGY_coptic_pop_cap min = 0.01 max = 1 }
}

# Gives a flow of relative party popularity to whichever party currently leads within
# [coptic_range_start, coptic_range_end) (used to auto-distribute the Coptic bloc's share
# among other non-Islamist outlooks when the ruling party is Islamist).
# Params (temp vars): coptic_range_start, coptic_range_end, coptic_range_flow
EGY_coptic_boost_leading_party_in_range = { # CKEdited
	set_temp_variable = { coptic_range_leader = coptic_range_start }
	set_temp_variable = { coptic_range_leader_pop = party_pop_array^coptic_range_start }
	for_loop_effect = {
		start = coptic_range_start
		end = coptic_range_end
		value = coptic_range_i
		if = {
			limit = { check_variable = { party_pop_array^coptic_range_i > coptic_range_leader_pop } }
			set_temp_variable = { coptic_range_leader = coptic_range_i }
			set_temp_variable = { coptic_range_leader_pop = party_pop_array^coptic_range_i }
		}
	}
	set_temp_variable = { party_index = coptic_range_leader }
	set_temp_variable = { party_popularity_increase = coptic_range_flow }
	set_temp_variable = { temp_outlook_increase = coptic_range_flow }
	change_relative_party_popularity = yes
}

EGY_coptic_monthly_politics_update = { # CKEdited
	EGY_calculate_coptic_pop_cap = yes

	# Islamist ruling party check: Freedom and Justice (Muslim Brotherhood), Hazemoon,
	# Salafist (Kingdom/Caliphate), Shiite (Vilayat_e_Faqih/Mod_Vilayat_e_Faqih)
	set_variable = { EGY_coptic_islamist_ruling = 0 }
	if = {
		limit = {
			OR = {
				is_in_array = { ruling_party = 8 }
				is_in_array = { ruling_party = 9 }
				is_in_array = { ruling_party = 10 }
				is_in_array = { ruling_party = 11 }
				is_in_array = { ruling_party = 12 }
				is_in_array = { ruling_party = 14 }
			}
		}
		set_variable = { EGY_coptic_islamist_ruling = 1 }
	}

	# Satisfaction-tier split of the Coptic bloc's political weight (ruling party / self), out of 100.
	# Mirrors the existing 5-tier satisfaction ladder thresholds used by update_coptic_ideas.
	set_variable = { EGY_coptic_share_ruling = 30 }
	set_variable = { EGY_coptic_share_self = 40 }
	if = {
		limit = { check_variable = { EGY_coptic_people_opinion > 79 } } # Very Satisfied
		set_variable = { EGY_coptic_share_ruling = 75 }
		set_variable = { EGY_coptic_share_self = 10 }
	}
	else_if = {
		limit = { check_variable = { EGY_coptic_people_opinion > 59 } } # Satisfied
		set_variable = { EGY_coptic_share_ruling = 55 }
		set_variable = { EGY_coptic_share_self = 20 }
	}
	else_if = {
		limit = { check_variable = { EGY_coptic_people_opinion > 49 } } # Neutral
		set_variable = { EGY_coptic_share_ruling = 30 }
		set_variable = { EGY_coptic_share_self = 40 }
	}
	else_if = {
		limit = { check_variable = { EGY_coptic_people_opinion > 29 } } # Not Satisfied
		set_variable = { EGY_coptic_share_ruling = 10 }
		set_variable = { EGY_coptic_share_self = 60 }
	}
	else = { # Enraged
		set_variable = { EGY_coptic_share_ruling = 0 }
		set_variable = { EGY_coptic_share_self = 85 }
	}

	set_variable = { EGY_coptic_share_other = 100 }
	subtract_from_variable = { EGY_coptic_share_other = EGY_coptic_share_ruling }
	subtract_from_variable = { EGY_coptic_share_other = EGY_coptic_share_self }

	# An Islamist ruling party never benefits - redirect its would-be share to other outlooks instead
	if = {
		limit = { check_variable = { EGY_coptic_islamist_ruling = 1 } }
		add_to_variable = { EGY_coptic_share_other = EGY_coptic_share_ruling }
		set_variable = { EGY_coptic_share_ruling = 0 }
	}

	set_temp_variable = { coptic_flow_base = 0.006 }

	# --- Coptic party's own growth, capped by state 217's population share unless radicalized (Enraged) ---
	if = {
		limit = {
			check_variable = { EGY_coptic_share_self > 0 }
			OR = {
				check_variable = { EGY_coptic_people_opinion < 30 }
				check_variable = { party_pop_array^13 < EGY_coptic_pop_cap }
			}
		}
		set_temp_variable = { coptic_flow_self = coptic_flow_base }
		multiply_temp_variable = { coptic_flow_self = EGY_coptic_share_self }
		divide_temp_variable = { coptic_flow_self = 100 }
		set_temp_variable = { party_index = 13 }
		set_temp_variable = { party_popularity_increase = coptic_flow_self }
		set_temp_variable = { temp_outlook_increase = coptic_flow_self }
		change_relative_party_popularity = yes
	}

	# --- Reward the ruling party when Copts are satisfied and it isn't Islamist ---
	if = {
		limit = { check_variable = { EGY_coptic_share_ruling > 0 } }
		set_temp_variable = { coptic_flow_ruling = coptic_flow_base }
		multiply_temp_variable = { coptic_flow_ruling = EGY_coptic_share_ruling }
		divide_temp_variable = { coptic_flow_ruling = 100 }
		set_party_index_to_ruling_party = yes
		set_temp_variable = { party_popularity_increase = coptic_flow_ruling }
		set_temp_variable = { temp_outlook_increase = coptic_flow_ruling }
		change_relative_party_popularity = yes
	}

	# --- Remaining share: spread across the leading non-Islamist party in each other outlook ---
	if = {
		limit = { check_variable = { EGY_coptic_share_other > 0 } }
		set_temp_variable = { coptic_other_flow = coptic_flow_base }
		multiply_temp_variable = { coptic_other_flow = EGY_coptic_share_other }
		divide_temp_variable = { coptic_other_flow = 100 }
		divide_temp_variable = { coptic_other_flow = 4 }

		set_temp_variable = { coptic_range_flow = coptic_other_flow }
		set_temp_variable = { coptic_range_start = 0 } # democratic
		set_temp_variable = { coptic_range_end = 4 }
		EGY_coptic_boost_leading_party_in_range = yes

		set_temp_variable = { coptic_range_flow = coptic_other_flow }
		set_temp_variable = { coptic_range_start = 4 } # communism, excluding Shiite (8, 9)
		set_temp_variable = { coptic_range_end = 8 }
		EGY_coptic_boost_leading_party_in_range = yes

		set_temp_variable = { coptic_range_flow = coptic_other_flow }
		set_temp_variable = { coptic_range_start = 15 } # neutrality, excluding Muslim Brotherhood (12), Coptic (13), Hazemoon (14)
		set_temp_variable = { coptic_range_end = 20 }
		EGY_coptic_boost_leading_party_in_range = yes

		set_temp_variable = { coptic_range_flow = coptic_other_flow }
		set_temp_variable = { coptic_range_start = 20 } # nationalist
		set_temp_variable = { coptic_range_end = 24 }
		EGY_coptic_boost_leading_party_in_range = yes
	}
}
