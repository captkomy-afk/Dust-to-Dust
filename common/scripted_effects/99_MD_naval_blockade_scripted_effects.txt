# Generic naval blockade system - common/scripted_diplomatic_actions/99_MD_naval_blockade_diplomatic_actions.txt.
# Any country (ROOT) can attempt to blockade any other (target). Effectiveness depends on whether
# ROOT holds naval dominance over every SEA strategic region touching the target's own coastline -
# full strength if ROOT controls all of them, partial (reduced target penalty, a political-power
# cost to ROOT) otherwise.
#
# A state's own `region` trigger returns its LAND strategic region (e.g. Israel's coast is all in
# region 174 "Levant"), never the adjacent sea region - HOI4 has no scripted way to discover
# province-level sea adjacency (no coordinate data, no adjacency table beyond the handful of
# explicit canal/strait rows in map/adjacencies.csv). Checking has_naval_control against a state's
# own land region is meaningless if that region has no water of its own: verified every one of
# Levant's provinces is `land`/`lake` type in map/definition.csv (zero `sea`), same for strategic
# region 212 "Cuba" (see common/scripted_triggers/99_USA_CUB_scripted_triggers.txt, which had the
# identical bug - has_naval_control = 212 was permanently unsatisfiable and is now removed there).
#
# The fix is a hand-curated LAND region -> SEA region(s) map, built once from the 108 land and 112
# sea strategic regions' names/geography (same manual-curation approach as the existing
# common/decisions/netherlands.txt VOC content, which hardcodes region IDs per-country - just done
# once per region instead of once per country, so it covers every country automatically and keeps
# working as borders change). Every one of the 112 sea regions also includes ITSELF as a candidate
# alongside its mapped land neighbors, since some strategic regions are mixed (e.g. 170 "Florida
# Coast" contains both Florida's land provinces and the sea province between Florida and the
# Bahamas in one ID) - a country whose own state sits directly inside a sea-flagged region needs
# that same ID checked, not just the land regions curated as bordering it.
#
# MD_naval_blockade_count_target_regions below is keyed by SEA region: for each of the 112 sea
# regions, it checks whether the target owns a state in that region OR any of its mapped land
# neighbors (relevant), and if so whether ROOT holds has_naval_control of that SEA region
# specifically (controlled) - never a land region. A land region bordering multiple seas (Levant ->
# Eastern Mediterranean AND Red Sea, via Tel Aviv and Eilat respectively) contributes to each; a
# sea bordered by multiple land regions (e.g. Eastern Mediterranean -> Greece, Syria, Egypt, Turkey,
# Levant) is only counted once per target regardless of how many of those land regions they own,
# since the OR collapses it to one relevant/controlled pair. This hand-built map is not guaranteed
# exhaustive or pixel-perfect - it's geography-name-based curation, not derived from province data,
# so a country whose coastline turns out to sit in an unmapped corner will read as having zero
# relevant regions (safe fallback: blockade has no effect on them rather than a bogus full/partial
# result). Flag any misses for a map fix the same way the Levant and Florida Coast gaps were found -
# trace the state's provinces to their strategic_region file, then check whether that region (or,
# for a mixed region, the region's own ID) ever appears in this file's OR blocks.
#
# Effect: MD_naval_blockade_count_target_regions

# target either sits inside directly or borders via a mapped land region) and

# Effect: MD_naval_blockade_recompute_for_target
# Purpose: Wraps MD_naval_blockade_count_target_regions (same calling convention) with the
# full/partial decision. Full requires ROOT to hold naval control of every one of the target's
# relevant regions (100%, not a majority) - a target with zero relevant regions (landlocked, or an
# authoring gap) degrades safely to partial rather than a bogus "full" blockade. Below 100%, every
# field of MD_naval_blockade_denied_modifier_partial is set to MD_naval_blockade_denied_idea's full
# magnitude times the live control ratio (MD_naval_blockade_this_ratio), so a near-complete
# blockade bites almost as hard as a full one instead of a single fixed half-strength value. Also
# feeds this target's ratio into ROOT.MD_naval_blockade_worst_ratio (the caller,
# MD_naval_blockade_weekly_recompute, resets it to 1 before the loop and uses the post-loop minimum
# to scale ROOT's own understrength political-power penalty).
MD_naval_blockade_recompute_for_target = {
	MD_naval_blockade_count_target_regions = yes
	if = {
		limit = {
			check_variable = { MD_naval_blockade_controlled_regions > 0 }
			NOT = { check_variable = { MD_naval_blockade_controlled_regions < MD_naval_blockade_relevant_regions } }
		}
		set_variable = { MD_naval_blockade_this_ratio = 1 }
		if = { limit = { has_dynamic_modifier = { modifier = MD_naval_blockade_denied_modifier_partial } }
			remove_dynamic_modifier = { modifier = MD_naval_blockade_denied_modifier_partial }
		}
		if = { limit = { NOT = { has_idea = MD_naval_blockade_denied_idea } } add_ideas = MD_naval_blockade_denied_idea }
	}
	else = {
		set_variable = {
			MD_naval_blockade_this_ratio = {
				value = MD_naval_blockade_controlled_regions
				divide = { value = MD_naval_blockade_relevant_regions clamp = { min = 0.001 max = 1000000 } }
				clamp = { min = 0 max = 1 }
			}
		}
		if = { limit = { has_idea = MD_naval_blockade_denied_idea } remove_ideas = MD_naval_blockade_denied_idea }
		set_variable = { MD_naval_blockade_partial_pop_tax = { value = MD_naval_blockade_this_ratio multiply = -0.15 } }
		set_variable = { MD_naval_blockade_partial_agri = { value = MD_naval_blockade_this_ratio multiply = -0.25 } }
		set_variable = { MD_naval_blockade_partial_pop = { value = MD_naval_blockade_this_ratio multiply = -0.02 } }
		set_variable = { MD_naval_blockade_partial_oil = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_aluminium = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_rubber = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_tungsten = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_steel = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_chromium = { value = MD_naval_blockade_this_ratio multiply = -0.10 } }
		set_variable = { MD_naval_blockade_partial_ind_cap = { value = MD_naval_blockade_this_ratio multiply = -0.2 } }
		set_variable = { MD_naval_blockade_partial_build_speed = { value = MD_naval_blockade_this_ratio multiply = -0.2 } }
		set_variable = { MD_naval_blockade_partial_pp = { value = MD_naval_blockade_this_ratio multiply = -0.03 } }

		set_variable = { MD_naval_blockade_partial_max_eff = { value = MD_naval_blockade_this_ratio multiply = -0.30 } }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = MD_naval_blockade_denied_modifier_partial } } }
			add_dynamic_modifier = { modifier = MD_naval_blockade_denied_modifier_partial }

	}
	ROOT = {
		set_variable = { MD_naval_blockade_worst_ratio = { value = MD_naval_blockade_worst_ratio min = PREV.MD_naval_blockade_this_ratio } }
	}
}

# Effect: MD_naval_blockade_weekly_recompute
# Purpose: Called from ROOT's own complete_effect on impose/lift (immediate feedback) and every
# week from the global on_weekly (common/on_actions/MD_on_actions.txt, gated behind
# MD_naval_blockade_targets^num > 0 so it's a cheap no-op for the countries not currently
# blockading anyone) while ROOT is blockading at least one target. Walks every one of ROOT's
# active blockades, recomputing full/partial and the target's modifier fresh each time rather than
# caching a per-pair mode - simplest way to also find ROOT's WORST current control ratio (the
# floor across every target) for ROOT's own understrength-blockade political power penalty, which
# scales as -10% * (1 - worst ratio): a near-complete worst blockade costs little, a token one
# costs close to the full -10%. At the cost of a fresh 112-region tally per target per week
# (bounded and rare - only active blockaders pay it). If two different countries are simultaneously
# blockading the SAME target, the target's modifier reflects whichever blockader's recompute ran
# most recently that week (last write wins) - an accepted simplification for what should be a rare
# multi-blockader edge case.
MD_naval_blockade_weekly_recompute = {
	set_variable = { MD_naval_blockade_worst_ratio = 1 }
	for_each_scope_loop = {
		array = MD_naval_blockade_targets
		MD_naval_blockade_recompute_for_target = yes
	}
	set_variable = {
		MD_naval_blockade_understrength_pp = {
			value = MD_naval_blockade_worst_ratio
			subtract = 1
			multiply = 0.10
		}
	}
	if = {
		limit = { check_variable = { MD_naval_blockade_worst_ratio < 1 } }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = MD_naval_blockade_understrength_modifier } } }
			add_dynamic_modifier = { modifier = MD_naval_blockade_understrength_modifier }
		}
	}
	else = {
		if = { limit = { has_dynamic_modifier = { modifier = MD_naval_blockade_understrength_modifier } }
			remove_dynamic_modifier = { modifier = MD_naval_blockade_understrength_modifier }
		}
	}
}
