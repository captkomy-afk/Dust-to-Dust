# Maritime chokepoint per-country blockade actions.
# Sent from the controlling country (ROOT) to a specific target country, mirroring the
# mt_sanctions_diplomatic_actions.txt ROOT/PREV bookkeeping pattern. complete_effect runs
# in the target's own scope, so setting a country_flag there directly feeds the matching
# is_disabled check in map/adjacency_rules.txt (evaluated for the transiting country).
# ROOT is always required to be human (is_ai = no) - AI cannot initiate these.
scripted_diplomatic_actions = {

	### SUEZ CANAL (EGY) ###

	MD_suez_block_impose = {
		icon = 19
		cost = 25

		visible = {
			ROOT = {
				is_ai = no
				OR = { controls_state = 213 controls_state = 214 }
			}
			NOT = { is_in_array = { array = MD_suez_block_on_us value = ROOT } }
		}

		send_description = MD_SUEZ_BLOCK_IMPOSE_SEND
		accept_title = MD_SUEZ_BLOCK_IMPOSE_ACCEPT
		accept_description = MD_SUEZ_BLOCK_IMPOSE_ACCEPT_DESC

		requires_acceptance = no
		complete_effect = {
			set_country_flag = SUEZ_CANAL_BLOCKED_FOR_COUNTRY
			add_ideas = MD_suez_denied_idea
			add_to_array = { array = MD_suez_block_on_us value = ROOT }
			add_opinion_modifier = { target = ROOT modifier = MD_Chokepoint_Blockade }
			add_opinion_modifier = { target = ROOT modifier = MD_Chokepoint_Blockade_Trade }
			# Wargoal isn't granted immediately - see MD_maritime_chokepoints.1
			# (events/MD_maritime_chokepoints.txt), which fires 10 days from now and grants it
			# (to us and to any other current Suez controller besides the blocker) only if we're
			# still blocked and not exempted by then, giving the blocker time to walk it back.
			country_event = { id = MD_maritime_chokepoints.1 days = 10 }
			ROOT = {
				add_to_array = { array = MD_suez_block_by_us value = PREV }
			}
		}

		ai_desire = { base = 0 }
	}

	MD_suez_block_lift = {
		icon = 19
		cost = 10

		visible = {
			is_in_array = { array = MD_suez_block_on_us value = ROOT }
		}

		send_description = MD_SUEZ_BLOCK_LIFT_SEND
		accept_title = MD_SUEZ_BLOCK_LIFT_ACCEPT
		accept_description = MD_SUEZ_BLOCK_LIFT_ACCEPT_DESC

		requires_acceptance = no
		complete_effect = {
			clr_country_flag = SUEZ_CANAL_BLOCKED_FOR_COUNTRY
			remove_ideas = MD_suez_denied_idea
			remove_from_array = { array = MD_suez_block_on_us value = ROOT }
			remove_opinion_modifier = { target = ROOT modifier = MD_Chokepoint_Blockade }
			remove_opinion_modifier = { target = ROOT modifier = MD_Chokepoint_Blockade_Trade }
			ROOT = {
				remove_from_array = { array = MD_suez_block_by_us value = PREV }
